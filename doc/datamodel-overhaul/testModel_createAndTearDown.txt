-- Stand: entspricht dem Ziel-Datenmodell aus datamodel.md (Abschnitte 3-11).
-- Eingearbeitet: snake_case, FK-Spalten varchar(18), durchgaengig ON DELETE
-- RESTRICT, typfreie node-Tabelle mit cover_node_id/reversed, app_node mit
-- relation + Wildcard (app_id IS NULL), content_item mit type-CHECK,
-- legacy_id fuer Deep-Links, Indizes auf allen FK-Spalten.
--

--
-- Name: generate_custom_id(character varying, integer); Type: FUNCTION; Schema: public; Owner: walthdav
--

CREATE OR REPLACE FUNCTION public.generate_custom_id(objectprefix character varying, recordnumber integer) RETURNS character varying
    LANGUAGE plpgsql
    AS $$
DECLARE
    custom_id VARCHAR(18);
BEGIN
    -- Ensure objectPrefix is exactly 4 characters
    IF LENGTH(objectPrefix) != 4 THEN
        RAISE EXCEPTION 'objectPrefix must be exactly 4 characters';
    END IF;

    -- Generate the custom ID
    custom_id := objectPrefix || LPAD(recordNumber::TEXT, 14, '0');

    -- Ensure the custom ID is exactly 18 characters
    IF LENGTH(custom_id) != 18 THEN
        RAISE EXCEPTION 'Generated ID must be exactly 18 characters';
    END IF;

    RETURN custom_id;
END;
$$;


-- ALTER FUNCTION public.generate_custom_id(objectprefix character varying, recordnumber integer) OWNER TO walthdav;

--
-- Name: set_table_id(); Type: FUNCTION; Schema: public; Owner: walthdav
--

CREATE OR REPLACE FUNCTION public.set_table_id() RETURNS trigger
    LANGUAGE plpgsql
    AS $$
DECLARE
    table_prefix TEXT;
BEGIN
    -- Get the prefix for the current table
    SELECT prefix INTO table_prefix
    FROM table_prefixes
    WHERE table_name = TG_TABLE_NAME;

    -- If no prefix found, use a default
    IF table_prefix IS NULL THEN
        table_prefix := '000x';
    END IF;

    NEW.id := generate_custom_id(table_prefix, NEW.recordnumber);
    RETURN NEW;
END;
$$;

------------------
-- Create Tables
------------------
--
-- Die Beziehungen werden bewusst erst weiter unten per ALTER TABLE ergaenzt:
-- content_node.active_content_item und content_item.content_node_id zeigen
-- aufeinander, eine der beiden Tabellen existiert beim CREATE also noch nicht.
--

-- name traegt den Wert aus APPLICATION_APPLICATION_KEY und ist damit der
-- Lookup-Schluessel der App, kein Anzeigename -> UNIQUE NOT NULL. Ein
-- Umbenennen ist eine Migration (Env-Dateien, Cache-Praefixe), kein Edit.
CREATE TABLE app (
  id            varchar(18) PRIMARY KEY NOT NULL,
  name          text UNIQUE NOT NULL,
  recordnumber  serial NOT NULL,
  createddate   timestamp without time zone DEFAULT CURRENT_TIMESTAMP NOT NULL
);

-- Typfrei: keine type/level-Spalte. cover_node_id und reversed sind optionale
-- Eigenschaften JEDES Knotens (frueher story.coverid bzw. chapter.reversed).
CREATE TABLE node (
  id                            varchar(18) PRIMARY KEY NOT NULL,
  name                          text,
  is_parent_controls_visibility boolean,
  sortnumber                    integer,
  reversed                      boolean,
  legacy_id                     varchar(18) UNIQUE,   -- alte 000s…/000c…-Id (Deep-Links)
  recordnumber                  serial NOT NULL,
  createddate                   timestamp without time zone DEFAULT CURRENT_TIMESTAMP NOT NULL,
  published_date                timestamp without time zone
);

-- app_id IS NULL = Wildcard (gilt fuer alle Apps, auch kuenftige).
CREATE TABLE app_node (
  id            varchar(18) PRIMARY KEY NOT NULL,
  relation      text NOT NULL CHECK (relation IN ('include', 'exclude')),
  recordnumber  serial NOT NULL,
  createddate   timestamp without time zone DEFAULT CURRENT_TIMESTAMP NOT NULL
);

-- Kein is_parent_controls_visibility: die Sichtbarkeit kontrolliert immer der node.
CREATE TABLE content_node (
  id              varchar(18) PRIMARY KEY NOT NULL,
  name            text,
  sortnumber      integer,
  legacy_id       varchar(18) UNIQUE,   -- alte 000p…-Id (Deep-Links)
  recordnumber    serial NOT NULL,
  createddate     timestamp without time zone DEFAULT CURRENT_TIMESTAMP NOT NULL,
  published_date  timestamp without time zone
);

-- Reiner Payload: keine eigene Sichtbarkeits-/Publish-Spalte.
CREATE TABLE content_item (
  id            varchar(18) PRIMARY KEY NOT NULL,
  content       text,
  type          text CHECK (type IN ('text', 'html', 'markdown', 'mermaid')),
  recordnumber  serial NOT NULL,
  createddate   timestamp without time zone DEFAULT CURRENT_TIMESTAMP NOT NULL
);

--  Create Relationships
------------------
--
-- Durchgaengig ON DELETE RESTRICT: nichts kaskadiert, geloescht wird bottom-up.
--

ALTER TABLE node
  ADD COLUMN parent_node_id varchar(18) REFERENCES node(id) ON DELETE RESTRICT;
ALTER TABLE node
  ADD COLUMN cover_node_id  varchar(18) REFERENCES node(id) ON DELETE RESTRICT;

ALTER TABLE app_node
  ADD COLUMN node_id varchar(18) NOT NULL REFERENCES node(id) ON DELETE RESTRICT;
ALTER TABLE app_node
  ADD COLUMN app_id  varchar(18) REFERENCES app(id) ON DELETE RESTRICT;

ALTER TABLE content_node
  ADD COLUMN node_id varchar(18) REFERENCES node(id) ON DELETE RESTRICT;
-- DEFERRABLE wegen des zirkulaeren Zeigers content_node <-> content_item:
-- der Zeiger darf innerhalb einer Transaktion vor dem Item gesetzt werden.
--
-- Gotcha: DEFERRABLE verschiebt nur die Existenzpruefung beim INSERT/UPDATE.
-- ON DELETE RESTRICT ist laut Postgres NICHT aufschiebbar und greift sofort -
-- ein content_item laesst sich also erst loeschen, wenn dieser Zeiger vorher
-- genullt oder umgebogen wurde (siehe Loeschreihenfolge unten). Wer das
-- Loeschen und das Nullen in derselben Transaktion in beliebiger Reihenfolge
-- erlauben will, nimmt hier ON DELETE NO ACTION - blockiert genauso, wird aber
-- erst beim COMMIT geprueft.
ALTER TABLE content_node
  ADD COLUMN active_content_item varchar(18)
    REFERENCES content_item(id) ON DELETE RESTRICT DEFERRABLE INITIALLY DEFERRED;

ALTER TABLE content_item
  ADD COLUMN content_node_id varchar(18) REFERENCES content_node(id) ON DELETE RESTRICT;

-- Solange es keine Versionierung gibt: je content_node hoechstens ein Item pro type.
ALTER TABLE content_item
  ADD CONSTRAINT content_item_node_type_unique UNIQUE (content_node_id, type);

------------------
-- Create Testdata
------------------
--
-- Aufbau (deckt die Entscheidungen aus datamodel.md ab):
--
--   app A "testAppA"        app B "testAppB"
--
--   root ................... include-Wildcard (app_id IS NULL) -> in A und B
--   |                        cover_node_id -> cover, reversed = false
--   +- cover ............... nicht veroeffentlicht (published_date IS NULL)
--   +- branch .............. erbt (is_parent_controls_visibility = true)
--   |  |                     reversed = true
--   |  +- leaf ............. dritte Ebene: zeigt die freie Verschachtelung
--   +- appAOnly ............ bricht die Kette (false) + eigener include fuer A
--   +- hiddenInB ........... erbt, aber exclude fuer B
--
--   branch traegt einen content_node mit zwei Repraesentationen (text/html);
--   active_content_item zeigt auf die html-Zeile.
--

DO $$
DECLARE
  v_app_a       varchar(18);
  v_app_b       varchar(18);
  v_root        varchar(18);
  v_cover       varchar(18);
  v_branch      varchar(18);
  v_leaf        varchar(18);
  v_app_a_only  varchar(18);
  v_hidden_in_b varchar(18);
  v_cnode       varchar(18);
  v_item_text   varchar(18);
  v_item_html   varchar(18);
BEGIN
  ---------------- apps ----------------
  INSERT INTO app (name) VALUES ('testAppA') RETURNING id INTO v_app_a;
  INSERT INTO app (name) VALUES ('testAppB') RETURNING id INTO v_app_b;

  ---------------- nodes ----------------
  INSERT INTO node (name, sortnumber, published_date, legacy_id)
    VALUES ('root', 10, CURRENT_TIMESTAMP, '000s00000000000011')
    RETURNING id INTO v_root;

  -- Cover-Knoten: bewusst unveroeffentlicht, dient nur als Bild-/Cover-Ziel.
  INSERT INTO node (name, sortnumber, parent_node_id, is_parent_controls_visibility)
    VALUES ('cover', 10, v_root, true)
    RETURNING id INTO v_cover;

  UPDATE node SET cover_node_id = v_cover, reversed = false WHERE id = v_root;

  INSERT INTO node (name, sortnumber, parent_node_id, is_parent_controls_visibility, reversed, published_date, legacy_id)
    VALUES ('branch', 20, v_root, true, true, CURRENT_TIMESTAMP, '000c00000000000045')
    RETURNING id INTO v_branch;

  -- Dritte Ebene - im alten Modell nicht darstellbar.
  INSERT INTO node (name, sortnumber, parent_node_id, is_parent_controls_visibility, published_date)
    VALUES ('leaf', 10, v_branch, true, CURRENT_TIMESTAMP)
    RETURNING id INTO v_leaf;

  -- Bricht die Vererbung (false) und haengt sich selbst nur an App A.
  INSERT INTO node (name, sortnumber, parent_node_id, is_parent_controls_visibility, published_date)
    VALUES ('appAOnly', 30, v_root, false, CURRENT_TIMESTAMP)
    RETURNING id INTO v_app_a_only;

  -- Erbt vom root, wird aber fuer App B ausgeschlossen.
  INSERT INTO node (name, sortnumber, parent_node_id, is_parent_controls_visibility, published_date)
    VALUES ('hiddenInB', 40, v_root, true, CURRENT_TIMESTAMP)
    RETURNING id INTO v_hidden_in_b;

  ---------------- app_node ----------------
  -- Wildcard: root gilt fuer alle Apps (loest die alte '*'-Konvention ab).
  INSERT INTO app_node (node_id, app_id, relation) VALUES (v_root,        NULL,    'include');
  -- Spezifisch: appAOnly haengt trotz gebrochener Kette an App A.
  INSERT INTO app_node (node_id, app_id, relation) VALUES (v_app_a_only,  v_app_a, 'include');
  -- Ausschluss: exclude schlaegt include.
  INSERT INTO app_node (node_id, app_id, relation) VALUES (v_hidden_in_b, v_app_b, 'exclude');

  ---------------- content ----------------
  INSERT INTO content_node (name, sortnumber, node_id, published_date, legacy_id)
    VALUES ('First ContentNode', 10, v_branch, CURRENT_TIMESTAMP, '000p00000000000007')
    RETURNING id INTO v_cnode;

  INSERT INTO content_item (content, type, content_node_id)
    VALUES ('Some random content', 'text', v_cnode)
    RETURNING id INTO v_item_text;

  INSERT INTO content_item (content, type, content_node_id)
    VALUES ('<p>Some random content</p>', 'html', v_cnode)
    RETURNING id INTO v_item_html;

  -- Explizite Auswahl statt des frueheren impliziten "html gewinnt, wenn gefuellt".
  UPDATE content_node SET active_content_item = v_item_html WHERE id = v_cnode;
END $$;

-- Baum von der Wurzel aus, mit Zyklus-Schutz (parent_node_id ist ein Selbstbezug):
WITH RECURSIVE tree AS (
  SELECT n.id, n.name, n.parent_node_id, 1 AS depth, ARRAY[n.id] AS path
  FROM node n
  WHERE n.parent_node_id IS NULL
  UNION ALL
  SELECT c.id, c.name, c.parent_node_id, t.depth + 1, t.path || c.id
  FROM node c
  JOIN tree t ON c.parent_node_id = t.id
  WHERE NOT c.id = ANY(t.path)
)
SELECT depth, repeat('  ', depth - 1) || name AS tree, id
FROM tree
ORDER BY path;
